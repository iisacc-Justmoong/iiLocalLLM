    if(IILOCALLLM_TEST_CHAT_GGUF)
        add_test(NAME iiLocalLLM.installed_tasks_inference COMMAND iiLocalLLMTaskInferenceConsumer "${IILOCALLLM_TEST_CHAT_GGUF}")

    endif()
    if(IILOCALLLM_TEST_AGENT_CATALOG)
        if(NOT IILOCALLLM_TEST_AGENT_MODEL MATCHES "^model://")
            message(FATAL_ERROR "Agent catalog tests require a model:// URI")
        endif()
        set(_task_profile_args)
        if(IILOCALLLM_TEST_AGENT_NO_THINK)
            list(APPEND _task_profile_args --no-think)
        endif()
        if(NOT IILOCALLLM_TEST_AGENT_TOOL_GRAMMAR)
            list(APPEND _task_profile_args --no-tool-grammar)
        endif()
        add_test(NAME iiLocalLLM.installed_tasks_catalog COMMAND iiLocalLLMTaskInferenceConsumer --catalog "${IILOCALLLM_TEST_AGENT_CATALOG}" "${IILOCALLLM_TEST_AGENT_MODEL}" ${_task_profile_args})
        add_test(NAME iiLocalLLM.installed_tasks_discovery COMMAND iiLocalLLMTaskInferenceConsumer --catalog "${IILOCALLLM_TEST_AGENT_CATALOG}" "${IILOCALLLM_TEST_AGENT_MODEL}" --deferred ${_task_profile_args})
        set_tests_properties(iiLocalLLM.installed_tasks_catalog iiLocalLLM.installed_tasks_discovery PROPERTIES TIMEOUT 600)
        if(IILOCALLLM_TEST_AGENT_THINKING_CONTROL)
            add_test(NAME iiLocalLLM.installed_tasks_thinking_control COMMAND iiLocalLLMTaskInferenceConsumer --catalog "${IILOCALLLM_TEST_AGENT_CATALOG}" "${IILOCALLLM_TEST_AGENT_MODEL}" --deferred --disable-thinking ${_task_profile_args})
            set_tests_properties(iiLocalLLM.installed_tasks_thinking_control PROPERTIES TIMEOUT 600)
        endif()
        if(UNIX AND NOT IOS AND NOT ANDROID AND NOT EMSCRIPTEN)
            add_executable(iiLocalLLMInputQueueInferenceConsumer ../input_queue_runtime_smoke.cpp)
            target_link_libraries(iiLocalLLMInputQueueInferenceConsumer PRIVATE iiLocalLLM::iiLocalLLM)

            set_tests_properties(iiLocalLLM.installed_input_queue_inference PROPERTIES TIMEOUT 600)
            add_executable(iiLocalLLMShellInferenceConsumer ../shell_runtime_smoke.cpp)
            target_link_libraries(iiLocalLLMShellInferenceConsumer PRIVATE iiLocalLLM::iiLocalLLM)
            add_test(NAME iiLocalLLM.installed_shell_inference COMMAND iiLocalLLMShellInferenceConsumer --catalog "${IILOCALLLM_TEST_AGENT_CATALOG}" "${IILOCALLLM_TEST_AGENT_MODEL}" ${_task_profile_args})
