cmake_minimum_required(VERSION 3.24)
project(iiLocalLLMConsumer LANGUAGES CXX)

find_package(iiLocalLLM 0.10.0 CONFIG REQUIRED)
add_executable(iiLocalLLMConsumer ../hello_world.cpp)
target_link_libraries(iiLocalLLMConsumer PRIVATE iiLocalLLM::iiLocalLLM)
set_target_properties(iiLocalLLMConsumer PROPERTIES CXX_EXTENSIONS OFF)
enable_testing()
if(UNIX AND NOT IOS AND NOT ANDROID AND NOT EMSCRIPTEN)
    add_executable(iiLocalLLMLocalApplicationsConsumer local_applications.cpp)
    target_link_libraries(iiLocalLLMLocalApplicationsConsumer PRIVATE iiLocalLLM::iiLocalLLM)
    add_test(NAME iiLocalLLM.installed_local_applications COMMAND iiLocalLLMLocalApplicationsConsumer)
    set_tests_properties(iiLocalLLM.installed_local_applications PROPERTIES TIMEOUT 30)
endif()
add_executable(iiLocalLLMDiscoveryConsumer discovery.cpp)
target_link_libraries(iiLocalLLMDiscoveryConsumer PRIVATE iiLocalLLM::iiLocalLLM)
add_test(NAME iiLocalLLM.installed_discovery_consumer COMMAND iiLocalLLMDiscoveryConsumer)
add_test(NAME iiLocalLLM.installed_consumer COMMAND iiLocalLLMConsumer)
add_executable(iiLocalLLMServiceConsumer consumer.cpp)
target_link_libraries(iiLocalLLMServiceConsumer PRIVATE iiLocalLLM::iiLocalLLM)
add_test(NAME iiLocalLLM.installed_service_consumer COMMAND iiLocalLLMServiceConsumer)
add_executable(iiLocalLLMAgentConsumer agent.cpp)
target_link_libraries(iiLocalLLMAgentConsumer PRIVATE iiLocalLLM::iiLocalLLM)
add_test(NAME iiLocalLLM.installed_agent_consumer COMMAND iiLocalLLMAgentConsumer)
add_executable(iiLocalLLMMcpServerConsumer mcp_server.cpp)
target_link_libraries(iiLocalLLMMcpServerConsumer PRIVATE iiLocalLLM::iiLocalLLM)
add_test(NAME iiLocalLLM.installed_mcp_server_consumer COMMAND iiLocalLLMMcpServerConsumer)
add_executable(iiLocalLLMMcpHttpServerConsumer mcp_http_server.cpp)
target_link_libraries(iiLocalLLMMcpHttpServerConsumer PRIVATE iiLocalLLM::iiLocalLLM)
add_test(NAME iiLocalLLM.installed_mcp_http_server_consumer COMMAND iiLocalLLMMcpHttpServerConsumer)
add_executable(iiLocalLLMAgentApiConsumer api.cpp)
target_link_libraries(iiLocalLLMAgentApiConsumer PRIVATE iiLocalLLM::iiLocalLLM)
add_test(NAME iiLocalLLM.installed_agent_api_consumer COMMAND iiLocalLLMAgentApiConsumer)
add_executable(iiLocalLLMCompactionConsumer compaction.cpp)
target_link_libraries(iiLocalLLMCompactionConsumer PRIVATE iiLocalLLM::iiLocalLLM)
add_test(NAME iiLocalLLM.installed_compaction_consumer COMMAND iiLocalLLMCompactionConsumer)
if(IILOCALLLM_TEST_CHAT_GGUF)
    if(IILOCALLLM_TEST_MCP_PYTHON)
        add_executable(iiLocalLLMDiscoveryInferenceConsumer ../agent_runtime_smoke.cpp)
        target_link_libraries(iiLocalLLMDiscoveryInferenceConsumer PRIVATE iiLocalLLM::iiLocalLLM)
        add_test(NAME iiLocalLLM.installed_configured_mcp_inference COMMAND iiLocalLLMDiscoveryInferenceConsumer "${IILOCALLLM_TEST_CHAT_GGUF}"
            "${IILOCALLLM_TEST_MCP_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/../mcp_official_server.py" --configured-eager)
        set_tests_properties(iiLocalLLM.installed_configured_mcp_inference PROPERTIES TIMEOUT 180)
        add_test(NAME iiLocalLLM.installed_discovery_inference COMMAND iiLocalLLMDiscoveryInferenceConsumer "${IILOCALLLM_TEST_CHAT_GGUF}"
            "${IILOCALLLM_TEST_MCP_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/../mcp_official_server.py" --discovery)
        set_tests_properties(iiLocalLLM.installed_discovery_inference PROPERTIES TIMEOUT 180)
    endif()
    add_executable(iiLocalLLMCompactionInferenceConsumer ../compaction_runtime_smoke.cpp)
    target_link_libraries(iiLocalLLMCompactionInferenceConsumer PRIVATE iiLocalLLM::iiLocalLLM)
    add_test(NAME iiLocalLLM.installed_compaction_inference COMMAND iiLocalLLMCompactionInferenceConsumer "${IILOCALLLM_TEST_CHAT_GGUF}")
    set_tests_properties(iiLocalLLM.installed_compaction_inference PROPERTIES TIMEOUT 240)

if(IILOCALLLM_TEST_MCP_PYTHON)
    add_executable(iiLocalLLMMcpConsumer ../mcp_official_smoke.cpp)
    target_link_libraries(iiLocalLLMMcpConsumer PRIVATE iiLocalLLM::iiLocalLLM)
    add_test(NAME iiLocalLLM.installed_mcp_consumer COMMAND iiLocalLLMMcpConsumer
        "${IILOCALLLM_TEST_MCP_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/../mcp_official_server.py")
    add_test(NAME iiLocalLLM.installed_mcp_http_consumer COMMAND iiLocalLLMMcpConsumer
        "${IILOCALLLM_TEST_MCP_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/../mcp_official_server.py" streamable-http)

    "TMPDIR=set:${CMAKE_CURRENT_BINARY_DIR}/tmp;TMP=set:${CMAKE_CURRENT_BINARY_DIR}/tmp;TEMP=set:${CMAKE_CURRENT_BINARY_DIR}/tmp;IILOCALLLM_APP_ENDPOINTS=set:${CMAKE_CURRENT_BINARY_DIR}/tmp/app-endpoints")
